                         🆘 EN CAS DE PROBLÈME
══════════════════════════════════════════════════════════════════════════

Problème: "port is already in use"
Solution:
  # Changer port dans docker-compose.yml
  ports:
    - "8080:80"   # External:Internal
  # Puis http://localhost:8080/


Problème: "Cannot connect to Database"
Solution:
  # Vérifier que service "db" démarre
  docker-compose logs db
  
  # Vérifier les credentials dans .env
  cat .env | grep MYSQL


Problème: "Volume mismatch"
Solution:
  # Vérifier chemin volume
  docker-compose config | grep "volumes:"
  # Devrait afficher: ".:/var/www/html"


Problème: "nginx/Apache won't start"
Solution:
  # Vérifier Dockerfile
  docker-compose logs app | tail -20
  # Chercher les erreurs Apache


Problème: "compositions start but app not running"
Solution:
  # Test du conteneur directement
  docker-compose exec app curl http://localhost/
  # Ou: docker-compose exec app /bin/bash
